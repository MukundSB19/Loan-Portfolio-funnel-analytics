// ---------------------------------------------------------------------
// 1. Query: Import_Loan_Applications
// Description: Ingests, cleans, type-casts, and enriches 25k+ applications
// ---------------------------------------------------------------------

let
    // 1. Source CSV connection (Update file path if moved)
    Source = Csv.Document(File.Contents("..\data\loan_applications_25k.csv"), [Delimiter=",", Columns=20, Encoding=65001, QuoteStyle=QuoteStyle.None]),
    
    // 2. Promote first row as Column Headers
    #"Promoted Headers" = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    
    // 3. Strict Data Type Transformations
    #"Changed Types" = Table.TransformColumnTypes(#"Promoted Headers", {
        {"application_id", type text},
        {"customer_id", type text},
        {"application_date", type date},
        {"channel", type text},
        {"applicant_age", Int64.Type},
        {"employment_type", type text},
        {"housing_status", type text},
        {"annual_income", Currency.Type},
        {"credit_score", Int64.Type},
        {"risk_tier", type text},
        {"debt_to_income_ratio", Percentage.Type},
        {"loan_purpose", type text},
        {"loan_amount", Currency.Type},
        {"loan_term_months", Int64.Type},
        {"interest_rate_pct", type number},
        {"final_stage_reached", type text},
        {"application_status", type text},
        {"portfolio_loan_status", type text},
        {"outstanding_balance", Currency.Type},
        {"total_funnel_duration_hours", type number}
    }),
    
    // 4. Add Calculated Columns for Rapid Filtering
    #"Added Is_Disbursed" = Table.AddColumn(#"Changed Types", "Is_Disbursed_Flag", each if [final_stage_reached] = "Disbursed" then 1 else 0, Int64.Type),
    #"Added DTI_Category" = Table.AddColumn(#"Added Is_Disbursed", "DTI_Category", each if [debt_to_income_ratio] < 0.20 then "1. Low (<20%)" else if [debt_to_income_ratio] < 0.35 then "2. Moderate (20-35%)" else if [debt_to_income_ratio] < 0.45 then "3. High (35-45%)" else "4. Critical (>45%)", type text)
in
    #"Added DTI_Category"
